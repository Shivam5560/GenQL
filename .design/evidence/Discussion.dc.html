SELECT DISTINCT
  c.c_customer_sk AS c_customer_sk,
  c.c_customer_id AS c_customer_id
FROM tpcds.store_sales AS ss
JOIN tpcds.customer_demographics AS cd
  ON cd.cd_credit_rating = 'Good'
  AND cd.cd_demo_sk = ss.ss_cdemo_sk
JOIN tpcds.customer AS c
  ON c.c_customer_sk = ss.ss_customer_sk
LIMIT 100
