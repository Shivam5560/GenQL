Finds customers who bought products through the Store channel but have a credit as good

Generated SQL Validated
SELECT DISTINCT
  c.c_customer_sk,
  c.c_customer_id
FROM tpcds.store_sales AS ss
JOIN tpcds.customer_demographics AS cd
  ON cd.cd_credit_rating = 'Good'
